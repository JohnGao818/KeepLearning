// 二叉树
typedef struct BiTNode
{
    int data;
    struct BiTNode *lchild, *rchild;
}BiTNode, *BiTree;

typedef struct LinkNode
{
    BiTree data;
    struct LinkNode *next;
}LinkNode;

typedef struct {
    LinkNode *front, *rear;
}*LinkQueue;

void InitQueue(LinkQueue &Q) {
    Q->front = Q->rear = (LinkNode*)malloc(sizeof(LinkNode));
    Q->front = NULL;
}

bool QueueEmpty(LinkQueue Q) {
    return Q->front == Q->rear;
}

bool EnQueue(LinkQueue &Q, BiTree x) {
    LinkNode *p = (LinkNode*)malloc(sizeof(LinkNode));
    p->data = x;
    Q->rear->next = p;
    Q->rear = p;
    p->next = NULL;
    return true;
}

bool DeQueue(LinkQueue &Q, BiTree x) {
    if (QueueEmpty(Q))
        return false;
    LinkNode *p = Q->front->next;
    x = p->data;
    Q->front->next = p->next;
    if (Q->rear == p)
        Q->rear = Q->front;
    free(p);
    return true;
}

bool GetHead(LinkQueue Q, BiTree x) {
    if (QueueEmpty(Q))
        return false;
    x = Q->front->data;
    return true;
}

typedef struct LinkNode
{
    BiTree data;
    struct LinkNode *next;
}*LiStack;

void InitStack(LiStack &S) {
    S = (LiStack)malloc(sizeof(LinkNode));
    S->next = NULL;
}

bool StackEmpty(LiStack S) {
    return S->next == NULL;
}

bool Push(LiStack &S, BiTree x) {
    LinkNode *p = (LinkNode*)malloc(sizeof(LinkNode));
    p->data = x;
    p->next = S->next;
    S->next = p;
    return true;
}

bool Pop(LiStack &S, BiTree x) {
    if (StackEmpty(S))
        return false;
    LinkNode *p = S->next;
    S->next = p->next;
    x = p->data;
    free(p);
    return true;
}

bool GetTop(LiStack &S, BiTree x) {
    if (StackEmpty(S))
        return false;
    x = S->next->data;
    return true;
}

void visit(BiTree T) {
    if (T != NULL)
        printf("%d", T->data);
}

// 先序遍历递归
void PreOrder(BiTree T) {
    if (T != NULL) {
        visit(T);
        PreOrder(T->lchild);
        PreOrder(T->rchild);
    }
}

// 先序遍历非递归
void PreOrder1(BiTree T) {
    LiStack S;
    InitStack(S);
    BiTree p = T;
    while (p || !StackEmpty(S)) {
        if (p) {
            visit(p);
            Push(S, p);
            p = p->lchild;
        } else {
            Pop(S, p);
            p = p->rchild;
        }
    }
    
}

// 中序遍历递归
void InOrder(BiTree T) {
    if (T != NULL) {
        PreOrder(T->lchild);
        visit(T);
        PreOrder(T->rchild);
    }
}
// 中序遍历非递归
void InOrder1(BiTree T) {
    LiStack S;
    InitStack(S);
    BiTree p = T;
    while (p || !StackEmpty(S)) {
        if (p) {
            Push(S, p);
            p = p->lchild;
        } else {
            Pop(S, p);
            visit(p);
            p = p->rchild;
        }
    }
    
}

// 后序遍历递归
void PostOrder(BiTree T) {
    if (T != NULL) {
        PreOrder(T->lchild);
        PreOrder(T->rchild);
        visit(T);
    }
}
// 后序遍历非递归
void PostOrder1(BiTree T) {
    LiStack S;
    InitStack(S);
    BiTree p = T, lastVisited;
    while (p || !StackEmpty(S)) {
        if (p) {
            Push(S, p);
            p = p->lchild;
        } else {
            GetTop(S, p);
            if (p->rchild != NULL && lastVisited != p->rchild) {
                Push(S, p->rchild);
                p = p->rchild->lchild;
            } else {
                Pop(S, p);
                visit(p);
                lastVisited = p;
                p = NULL;
            }
        }
    }
    
}

// 层次遍历
void LevelOrder(BiTree T) {
    LinkQueue Q;
    BiTree p;
    InitQueue(Q);
    EnQueue(Q, T);
    while(!QueueEmpty(Q)) {
        DeQueue(Q, p);
        visit(p);
        if (p->lchild != NULL)
            EnQueue(Q, p->lchild);
        if (p->rchild != NULL)
            EnQueue(Q, p->rchild);
    }
}

#### 线索二叉树

// 线索二叉树
typedef struct ThreadTNode
{
    int data;
    struct ThreadTNode *lchild, *rchild, *parent;
    int ltag, rtag;
}ThreadTNode, *ThreadTree;

void visit(ThreadTree T) {
    if (T != NULL)
        printf("%d", T->data);
}

// 递归构建中序线索二叉树
void InThread(ThreadTree p, ThreadTree pre) {
    if (p != NULL) {
        InThread(p->lchild, pre);
        if (p->lchild == NULL);
            p->lchild = pre;
            p->ltag = 1;
        if (pre != NULL && pre->rchild == NULL)
            pre->rchild = p;
            pre->rtag = 1;
        pre = p;
        InThread(p->rchild, pre);
    }
}

void CreateInThread(ThreadTree T) {
    ThreadTree pre = NULL;
    if (T != NULL) {
        InThread(T, pre);
        pre->rchild = NULL;
        pre->rtag = 1;
    }
}

// 中序线索二叉树的遍历
ThreadTree FirstNode(ThreadTree p) {
    while (p->ltag == 0)
        p = p->lchild;
    return p;
}

ThreadTree NextNode(ThreadTree p) {
    if (p->rtag == 1)
        return p->rchild;
    else
        return FirstNode(p->rchild);

}

void InOrder(ThreadTree T) {
    for (ThreadTree p = FirstNode(T); p != NULL; p = NextNode(p))
        visit(p);
}

// 递归构建先序线索二叉树
void PreThread(ThreadTree p, ThreadTree pre) {
    if (p != NULL) {
        if (p->lchild == NULL);
            p->lchild = pre;
            p->ltag = 1;
        if (pre != NULL && pre->rchild == NULL)
            pre->rchild = p;
            pre->rtag = 1;
        pre = p;
        InThread(p->lchild, pre);
        InThread(p->rchild, pre);
    }
}

void CreatePreThread(ThreadTree T) {
    ThreadTree pre = NULL;
    if (T != NULL) {
        InThread(T, pre);
        pre->rchild = NULL;
        pre->rtag = 1;
    }
}
// 先序线索二叉树的遍历
ThreadTree FirstNode(ThreadTree p) {
    if (p != NULL)
        return p;
}

ThreadTree NextNode(ThreadTree p) {
    if (p->ltag == 0 && p->lchild != NULL)
        return p->lchild;
    else
        return p->rchild;
}

void InOrder(ThreadTree T) {
    for (ThreadTree p = FirstNode(T); p != NULL; p = NextNode(p))
        visit(p);
}

// 递归构建后序线索二叉树
void PostThread(ThreadTree p, ThreadTree pre) {
    if (p != NULL) {
        InThread(p->lchild, pre);
        InThread(p->rchild, pre);
        if (p->lchild == NULL);
            p->lchild = pre;
            p->ltag = 1;
        if (pre != NULL && pre->rchild == NULL)
            pre->rchild = p;
            pre->rtag = 1;
        pre = p;
    }
}

void CreatePostThread(ThreadTree T) {
    ThreadTree pre = NULL;
    if (T != NULL) {
        InThread(T, pre);
        pre->rchild = NULL;
        pre->rtag = 1;
    }
}
// 后序线索二叉树的遍历
ThreadTree FirstNode(ThreadTree p) {
    while (p->ltag == 0)
        p = p->lchild;
    if (p->rtag == 0);
        return FirstNode(p->rchild);
    return p;
}

ThreadTree NextNode(ThreadTree p) {
    if (p == p->parent->rchild)
        return p->rchild;
    else if (p == p->parent->lchild && p->parent->rtag == 1)
        return p->rchild;
    else
        return FirstNode(p->parent->rchild);
}

void PostOrder(ThreadTree T) {
    for (ThreadTree p = FirstNode(T); p != NULL && p != T; p = NextNode(p))
        visit(p);
}
