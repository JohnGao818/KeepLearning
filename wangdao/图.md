// 无向图，边带权值
typedef struct
{
    VertexType vex[MaxVertexNum];
    EdgeType edge[MaxVertexNum][MaxVertexNum];
    int vexnum, edgenum;
} MGraph;

int FindVexIndex(MGraph G, char x) {
    for (int i = 0; i < G.vexnum; i++) {
        if (G.vex[i] == x) {
            return i;
        }
    }
    return -1;
}

bool Adjacent(MGraph G, char x, char y)
{
    int xIndex = FindVexIndex(G, x);
    int yIndex = FindVexIndex(G, y);
    if (xIndex == -1 || yIndex == -1)
        return false;
    return G.edge[xIndex][yIndex] > 0;
}

void Neighbors(MGraph G, char x)
{
    int xIndex = FindVexIndex(G, x);
    if (xIndex == -1)
        return;
    for(int i = 0; i < G.vexnum; i++) {
        if (G.edge[xIndex][i] > 0) {
            printf("%c - %c", x, G.vex[i]);
        }
    }
}

void AddEdge(MGraph G, char x, char y, int z)
{
    int xIndex = FindVexIndex(G, x);
    int yIndex = FindVexIndex(G, y);
    if (xIndex == -1 || yIndex == -1)
        return;
    G.edge[xIndex][yIndex] = z;
    G.edge[yIndex][xIndex] = z;
}

void RemoveEdge(MGraph G, char x, char y)
{
    int xIndex = FindVexIndex(G, x);
    int yIndex = FindVexIndex(G, y);
    if (xIndex == -1 || yIndex == -1)
        return;
    G.edge[xIndex][yIndex] = 0;
    G.edge[yIndex][xIndex] = 0;
}

int FirstNeighbor(MGraph G, char x)
{
    int xIndex = FindVexIndex(G, x);
    if (xIndex == -1)
        return -1;
    for (int i = 0; i < G.vexnum; i++)
        if (G.edge[xIndex][i] > 0)
            return i;
    return -1;
}

int NextNeighbor(MGraph G, int x, int y)
{
    if (x < 0)
        return -1;
    for (int i = y; i < G.vexnum; i++)
        if (G.edge[x][i] > 0)
            return i;
    return -1;
}
