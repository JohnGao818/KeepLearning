#### 直接插入排序
时间复杂度：O(n^2)
空间复杂度：O(1)
稳定性：稳定
```cpp
void InserSort(int a[], int n) {
    int i, j;
    for (i = 2; i < n + 1; i++) {
        a[0] = a[i];
        for (j = i -1; a[j] > a[0]; j--) {
            a[j + 1] = a[j];
        }
        a[j + 1] = a[0];
    }
}
```

#### 折半插入排序
空间复杂度：O(1)
时间复杂度：O(n^2)
稳定性：稳定
```cpp
void BinaryInserSort(int a[], int n) {
    for (int i = 2; i < n + 1; i++) {
        a[0] = a[i];
        int left = 1, right =  i - 1, mid;
        while (left <= right)
        {
            mid = (left + right) / 2;
            if (a[0] > a[mid]) {
                left = mid + 1;
            } else {
                right = mid - 1;
            }
        }
        for (int j = i; j > right; j--) {
            a[j] = a[j - 1];
        }
        a[right + 1] = a[0];
    }
}
```

#### 希尔排序
```cpp
void ShellSort(int a[], int n)
{
    int dk, i, j;
    for (dk = n/2; dk >= 1; dk = dk / 2) {
        for (i = dk + 1; i <= n; i++) {
            if (a[i] < a[i-dk]) {
                a[0] = a[i];
                for (j = i - dk; j > 0 && a[0] < a[j]; j-=dk) {
                    a[j+dk] = a[j];
                }
                a[j+dk] = a[0];
            }
        }
    }
}
```

#### 冒泡排序
```cpp
void BubbleSort(int a[], int n)
{
    int temp;
    for (int i = 1; i < n; i++)
    {
        for (int j = n; j > i; j--)
        {
            if (a[j] < a[j - 1])
            {
                temp = a[j - 1];
                a[j - 1] = a[j];
                a[j] = temp;
            }
        }
    }
}
```

#### 快速排序
```cpp
int Partition(int a[], int low, int high) {
    if (low < high) {
        a[0] = a[low];
        while (low < high)
        {
            while (a[high] >= a[0] && low < high)
            {
                high--;
            }
            a[low] = a[high];
            while (a[low] <= a[0] && low < high)
            {
                low++;
            }
            a[high] = a[low];
        }
        a[low] = a[0];
        return low;
    }
}

void QuickSort(int a[], int low, int high)
{
    if (low < high) {
        int pos = Partition(a, low, high);
        QuickSort(a, low, pos - 1);
        QuickSort(a, pos + 1, high);
    }
}
```

### 简单选择排序
```cpp
void SimpleSort(int a[], int n) {
    int temp, minIndex;
    for (int i = 1; i < n + 1; i++) {
        minIndex = i;
        for (int j = i + 1; j < n + 1; j++) {
            if (a[j] < a[minIndex]) {
                minIndex = j;
            }
        }
        temp = a[i];
        a[i] = a[minIndex];
        a[minIndex] = temp; 
    }
}
```

### 堆排序
```cpp
void HeapAdjust(int a[], int n, int x) {
    int leftChild = 2*x, rightChild = 2*x + 1, temp;
    if (x > n || leftChild  > n) {
        return;
    }
    if (a[leftChild] > a[x] && (rightChild > n || a[leftChild] > a[rightChild])) {
        temp = a[leftChild];
        a[leftChild] = a[x];
        a[x] = temp;
        HeapAdjust(a, n, leftChild);
    } else if (rightChild <= n && a[rightChild] > a[x] && a[rightChild] > a[leftChild]) {
        temp = a[rightChild];
        a[rightChild] = a[x];
        a[x] = temp;
        HeapAdjust(a, n, rightChild);
    }
}
void BuildMaxHeap(int a[], int n) {
    for(int i = n / 2; i > 0; i--) {
        HeapAdjust(a, n, i);
    }
}
void HeapSort(int a[], int n) {
    BuildMaxHeap(a, n);
    int temp;
    for(int i = n; i > 0; i--) {
        temp = a[1];
        a[1] = a[i];
        a[i] = temp;
        HeapAdjust(a, i-1, 1);
    }
}
```

#### 归并排序
```cpp
int B[9];
void Merge(int a[], int low, int mid, int high) {
    int i = low, j = low, k = mid+1;
    for (int i = low; i<=high; i++)
        B[i] = a[i];
    i = low;
    while (j<=mid && k<=high)
    {
        if (B[j] < B[k])
            a[i++] = B[j++];
        else
            a[i++] = B[k++];
    }
    while (j<=mid)
        a[i++] = B[j++];
    while (k<=high)
        a[i++] = B[k++];
}
void MergeSort(int a[], int low, int high) {
    if (low < high) {
        int mid = (high + low) / 2;
        MergeSort(a, low, mid);
        MergeSort(a, mid + 1, high);
        Merge(a, low, mid, high);
    }
}
```

```cpp
void CountSort(int a[], int b[], int n, int k) {
    int c[k];
    for(int i=0; i<k; i++)
        c[i] = 0;
    for(int i=1; i <= n; i++)
        c[a[i]] += 1;
    for(int i=0; i<k-1; i++)
        c[i+1] += c[i];
    for(int i=1; i <= n; i++){
        b[c[a[i]]] = a[i];
        c[a[i]] -= 1;
    }
}
```
